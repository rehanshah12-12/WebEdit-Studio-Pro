-- Run this file in Supabase Dashboard > SQL Editor.
-- These tables use Row Level Security so signed-in users can only access their own orders.
create table if not exists public.contact_requests (
  id bigint generated by default as identity primary key,
  name text not null,
  email text,
  message text not null,
  created_at timestamptz not null default now()
);

alter table public.contact_requests enable row level security;
-- No public SELECT policy is created. Contact submissions are inserted by the backend
-- using the anon key only if an INSERT policy is added; this starter intentionally
-- requires you to add a carefully designed policy or use a trusted server-only key.
-- Do not expose a service_role key in frontend code.

create table if not exists public.orders (
  id bigint generated by default as identity primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  service text not null,
  details text not null default '',
  status text not null default 'pending'
    check (status in ('pending', 'in_progress', 'completed', 'cancelled')),
  created_at timestamptz not null default now()
);

alter table public.orders enable row level security;

drop policy if exists "Users can view their own orders" on public.orders;
create policy "Users can view their own orders"
on public.orders for select
to authenticated
using (auth.uid() = user_id);

drop policy if exists "Users can create their own orders" on public.orders;
create policy "Users can create their own orders"
on public.orders for insert
to authenticated
with check (auth.uid() = user_id);

-- Contact submissions are intentionally not enabled by this SQL alone.
-- Before using /api/contact, choose a safe spam-control strategy and add an INSERT policy
-- or use a server-only Supabase service role key on the backend (never in public frontend).

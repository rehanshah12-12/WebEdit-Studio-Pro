document.addEventListener('DOMContentLoaded', () => { const filterButtons = document.querySelectorAll('[data-filter]'); const cards = document.querySelectorAll('[data-category]'); filterButtons.forEach(button => button.addEventListener('click', () => { const filter = button.getAttribute('data-filter'); filterButtons.forEach(b => b.classList.toggle('active-filter', b === button)); cards.forEach(card => { card.classList.toggle('hidden', filter !== 'all' && card.getAttribute('data-category') !== filter); }); })); });
